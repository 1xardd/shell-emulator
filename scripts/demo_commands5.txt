# Тестовый скрипт для этапа 5: команда rm.

# Смотрим исходное состояние
ls

# Удаляем файл
rm motd.txt
ls

# Пытаемся удалить тот же файл ещё раз — ошибка
rm motd.txt

# Пытаемся удалить непустую директорию — ошибка
rm home

# Заходим в глубокую папку
cd home
cd user
cd projects
ls

# Удаляем файл внутри
rm project1.txt
ls

# Удаляем пустую папку archive
rm archive
ls

# Возвращаемся в корень
cd /

# Удаляем пустую папку etc
rm etc
ls

# Проверяем работу с несколькими аргументами
rm /home/user/projects/project2.txt
ls /home/user/projects

# Проверяем ошибку при удалении корня
rm /

# Финальный ls
ls
