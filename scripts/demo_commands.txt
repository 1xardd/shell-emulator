# Это стартовый скрипт для эмулятора оболочки ОС.
# Строки, начинающиеся с решётки, — комментарии, они пропускаются.
# Пустые строки тоже пропускаются.

# Команда-заглушка ls без аргументов
ls

# Команда ls с аргументами
ls foo bar

# Команда ls с аргументом в кавычках (проверка парсера)
ls "моя папка"

# Команда ls с раскрытием переменной окружения
ls $HOME

# Команда cd с аргументом
cd docs

# Несуществующая команда — должна быть пропущена с предупреждением
abracadabra

# Проверка незакрытой кавычки — ошибка разбора, строка пропускается
ls "незакрытая

# Ещё одна команда, которая должна выполниться
ls final
