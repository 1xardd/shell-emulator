# Тестовый скрипт для проверки VFS (этап 3).
# Команды выполняются в окне эмулятора последовательно.

# Информация о VFS
vfs-info

# Смотрим корень
ls

# Заходим в home
cd home
ls

# Заходим в user
cd user
ls

# Заходим в projects
cd projects
ls

# Возвращаемся на уровень вверх
cd ..
ls

# Возвращаемся в корень
cd /
ls

# Проверяем работу с файлом (не директорией)
ls motd.txt

# Проверяем ошибку — несуществующий путь
ls /nonexistent

# Проверяем cd в несуществующую папку
cd missing

# Проверяем cd в файл (должна быть ошибка «не директория»)
cd motd.txt

# Возвращаемся в корень на всякий случай
cd /
